# The demo programs from issue #109, with the define correction, fixed and responsive.
# `== name` starts a program. Indentation and strips don't matter here: only the cards are counted.

== 1-fixed
[define(setup)]
[call(p5.createCanvas)][num{500}][num{500}]
[call(p5.background)][color(midnightblue)]
[call(p5.noStroke)]
[call(p5.fill)][color(gold)]
[repeat][num{30}]
  [set{x}to][call(random)][num{500}]
  [set{y}to][call(random)][num{500}]
  [call(p5.rect)][get{x}][get{y}][num{10}][num{16}]
[dedent]
[call(p5.fill)][color(hotpink)]
[repeat][num{30}]
  [set{x}to][call(random)][num{500}]
  [set{y}to][call(random)][num{500}]
  [call(p5.rect)][get{x}][get{y}][num{10}][num{16}]
[dedent]
[call(p5.fill)][color(aquamarine)]
[repeat][num{30}]
  [set{x}to][call(random)][num{500}]
  [set{y}to][call(random)][num{500}]
  [call(p5.rect)][get{x}][get{y}][num{10}][num{16}]
[dedent]

== 3-fixed
[define(setup)]
[call(p5.createCanvas)][num{400}][num{400}]
[call(p5.noStroke)]
[define(draw)]
[call(p5.background)][color(black)]
[call(p5.fill)][color(white)]
[set{tx}to][get(p5.frameCount)][*][num{0.015}]
[set{ty}to][get(p5.frameCount)][*][num{0.02}]
[for{x}in][call(range)][num{10}][num{400}][num{20}]
  [for{y}in][call(range)][num{10}][num{400}][num{20}]
    [set{nx}to][get{x}][*][num{0.015}][+][get{tx}]
    [set{ny}to][get{y}][*][num{0.02}][+][get{ty}]
    [set{n}to][call(p5.noise)][get{nx}][get{ny}]
    [call(p5.circle)][get{x}][get{y}][get{n}][*][num{20}]
  [dedent]
[dedent]

== 4-fixed
[global{xs}][call(list)]
[global{ys}][call(list)]
[global{fills}][call(list)]
[global{length}][num{30}]
[define(setup)]
[call(p5.createCanvas)][num{500}][num{500}]
[call(p5.noStroke)]
[for{c}in][call(list)][color(gold)][color(hotpink)][color(aquamarine)]
  [repeat][get{length}]
    [get{xs}][call(append)][call(random)][num{500}]
    [get{ys}][call(append)][call(random)][num{500}]
    [get{fills}][call(append)][get{c}]
  [dedent]
[dedent]
[define(draw)]
[call(p5.background)][color(midnightblue)]
[for{i}in][call(range)][get{length}][*][num{3}]
  [set{x}to][get{xs}][call(getAt)][get{i}]
  [set{y}to][get{ys}][call(getAt)][get{i}]
  [if][get{y}][>][num{500}]
    [set{y}to][num{-16}]
  [else]
    [set{y}to][get{y}][+][num{2}]
  [dedent]
  [get{ys}][call(setAt)][get{i}][get{y}]
  [call(p5.fill)][get{fills}][call(getAt)][get{i}]
  [call(p5.rect)][get{x}][get{y}][num{10}][num{16}]
[dedent]
